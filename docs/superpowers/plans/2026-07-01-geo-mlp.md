"""Spin(3)-equivariant grade dropout for Cl(3,0) multivectors.

Drops entire grade-k blocks independently. Each grade block is a Spin(3)
subrepresentation, so zeroing a block commutes with the rotor action ρ(w).
"""

import torch
import torch.nn.functional as F

from su2eqnn.primitives.linear import grade_project

_N_GRADES = 4
_MV_DIM = 8


def grade_dropout(x: torch.Tensor, p: float, training: bool = True) -> torch.Tensor:
    """Drop entire grade blocks of a Cl(3,0) multivector with probability p.

    Parameters
    ----------
    x : torch.Tensor with shape (..., 8)
    p : float
        Probability of dropping each grade block.
    training : bool
        If False, no dropout is applied (inference mode).

    Returns
    -------
    outputs : torch.Tensor with shape (..., 8)
    """
    x_grades = grade_project(x)                          # (..., 4, 8)
    batch_shape = x_grades.shape[:-2]
    h = x_grades.view(-1, _N_GRADES, _MV_DIM)           # (N, 4, 8)
    h = F.dropout1d(h, p=p, training=training, inplace=False)
    h = h.view(*batch_shape, _N_GRADES, _MV_DIM)        # (..., 4, 8)
    return h.sum(dim=-2)                                  # (..., 8)
```

import torch
from torch import nn

from su2eqnn.primitives.dropout import grade_dropout


class GradeDropout(nn.Module):
    """Equivariant dropout: drops grade blocks in MV stream, standard dropout in scalar stream.

    Parameters
    ----------
    p : float
        Dropout probability.
    """

    def __init__(self, p: float = 0.0) -> None:
        super().__init__()
        self._p = p

    def forward(
        self,
        multivectors: torch.Tensor,
        scalars: torch.Tensor,
    ) -> Tuple[torch.Tensor, torch.Tensor]:
        """Forward pass.

        Parameters
        ----------
        multivectors : torch.Tensor with shape (..., C, 8)
        scalars      : torch.Tensor with shape (..., S)

        Returns
        -------
        out_mv : torch.Tensor with shape (..., C, 8)
        out_s  : torch.Tensor with shape (..., S)
        """
        out_mv = grade_dropout(multivectors, p=self._p, training=self.training)
        out_s = torch.nn.functional.dropout(scalars, p=self._p, training=self.training)
        return out_mv, out_s

"""GeoMLP configuration."""

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, List, Mapping, Optional


@dataclass
class MLPConfig:
    """Geometric MLP configuration.

    Parameters
    ----------
    mv_channels : list of int
        Multivector channel sizes per layer, from input to output.
        Must have at least 2 entries.
    s_channels : list of int or None
        Scalar channel sizes per layer. If provided, must have same length as mv_channels.
    activation : {"relu", "sigmoid", "gelu"}
        Gated activation function for nonlinear layers.
    dropout_prob : float or None
        If set, applies GradeDropout after the first (GeometricBilinear) layer.
    """

    mv_channels: Optional[List[int]] = None
    s_channels: Optional[List[int]] = None
    activation: str = "gelu"
    dropout_prob: Optional[float] = None

    def __post_init__(self):
        if isinstance(self.dropout_prob, str) and self.dropout_prob.lower() in ("null", "none"):
            self.dropout_prob = None

    @classmethod
    def cast(cls, config: Any) -> MLPConfig:
        if isinstance(config, MLPConfig):
            return config
        if isinstance(config, Mapping):
            return cls(**config)
        raise ValueError(f"Cannot cast {config} to {cls}")
```

"""Scalar-gated nonlinearity for Cl(3,0) dual-stream (MV + scalar)."""

from typing import Tuple

import torch
from torch import nn

from su2eqnn.primitives.nonlinearities import gated_gelu, gated_relu, gated_sigmoid


class ScalarGatedNonlinearity(nn.Module):
    """Gated nonlinearity using grade-0 (scalar) component as gate.

    For MVs: gate = x[..., [0]] (grade-0, Spin(3)-invariant).
    Computes f(gate) * x — invariant scalar times equivariant MV = equivariant.

    For scalars: applies f pointwise (no gating needed).

    Parameters
    ----------
    nonlinearity : {"relu", "sigmoid", "gelu"}
    """

    def __init__(self, nonlinearity: str = "gelu", **kwargs) -> None:
        super().__init__()
        _gated = {"relu": gated_relu, "gelu": gated_gelu, "sigmoid": gated_sigmoid}
        _scalar = {
            "relu": nn.functional.relu,
            "gelu": nn.functional.gelu,
            "sigmoid": nn.functional.sigmoid,
        }
        if nonlinearity not in _gated:
            raise ValueError(f"Unknown nonlinearity '{nonlinearity}', options: {list(_gated)}")
        self.gated_fn = _gated[nonlinearity]
        self.scalar_fn = _scalar[nonlinearity]

    def forward(
        self,
        multivectors: torch.Tensor,
        scalars: torch.Tensor,
    ) -> Tuple[torch.Tensor, torch.Tensor]:
        """Forward pass.

        Parameters
        ----------
        multivectors : torch.Tensor with shape (..., C, 8)
        scalars      : torch.Tensor with shape (..., S)

        Returns
        -------
        out_mv : torch.Tensor with shape (..., C, 8)
        out_s  : torch.Tensor with shape (..., S)
        """
        gates = multivectors[..., [0]]              # (..., C, 1) — grade-0, invariant
        out_mv = self.gated_fn(multivectors, gates)
        out_s = self.scalar_fn(scalars)
        return out_mv, out_s
```

"""Spin(3)-equivariant geometric bilinear layer for Cl(3,0)."""

from typing import Optional, Tuple

import torch
from torch import nn

from su2eqnn.layers.layer_norm import EquiLayerNorm
from su2eqnn.layers.linear import EquiLinear
from su2eqnn.primitives.bilinear import geometric_product


class GeometricBilinear(nn.Module):
    """Geometric bilinear layer: projects to hidden, takes geometric product, projects out.

    The geometric product is the only Spin(3)-equivariant nonlinear operation:
      GP(ρ(w)(x), ρ(w)(y)) = ρ(w)(GP(x, y))

    Architecture:
      left  = linear_left(x, s)          # (..., hidden, 8)  standard init
      right = linear_right(x, s)         # (..., hidden, 8)  almost_unit_scalar init
      gp    = geometric_product(left, right)
      out_mv, out_s = linear_out(gp, s)
      out_mv = EquiLayerNorm(out_mv)

    linear_right uses almost_unit_scalar init so the GP starts near the identity
    map, stabilizing early training.

    Parameters
    ----------
    in_mv_channels : int
    out_mv_channels : int
    hidden_mv_channels : int or None
        Defaults to out_mv_channels.
    in_s_channels : int or None
    out_s_channels : int or None
    """

    def __init__(
        self,
        in_mv_channels: int,
        out_mv_channels: int,
        hidden_mv_channels: Optional[int] = None,
        in_s_channels: Optional[int] = None,
        out_s_channels: Optional[int] = None,
    ) -> None:
        super().__init__()

        if hidden_mv_channels is None:
            hidden_mv_channels = out_mv_channels

        self.linear_left = EquiLinear(
            in_mv_channels,
            hidden_mv_channels,
            in_s_channels=in_s_channels,
            out_s_channels=None,
        )
        # almost_unit_scalar requires in_s_channels; fall back to default when no scalars
        right_init = "almost_unit_scalar" if in_s_channels is not None else "default"
        self.linear_right = EquiLinear(
            in_mv_channels,
            hidden_mv_channels,
            in_s_channels=in_s_channels,
            out_s_channels=None,
            initialization=right_init,
        )
        self.linear_out = EquiLinear(
            hidden_mv_channels,
            out_mv_channels,
            in_s_channels=in_s_channels,
            out_s_channels=out_s_channels,
        )
        self.norm = EquiLayerNorm()

    def forward(
        self,
        multivectors: torch.Tensor,
        scalars: Optional[torch.Tensor] = None,
    ) -> Tuple[torch.Tensor, Optional[torch.Tensor]]:
        """Forward pass.

        Parameters
        ----------
        multivectors : torch.Tensor with shape (..., in_mv_channels, 8)
        scalars      : torch.Tensor with shape (..., in_s_channels) or None

        Returns
        -------
        out_mv : torch.Tensor with shape (..., out_mv_channels, 8)
        out_s  : torch.Tensor with shape (..., out_s_channels) or None
        """
        left, _  = self.linear_left(multivectors, scalars=scalars)
        right, _ = self.linear_right(multivectors, scalars=scalars)
        gp = geometric_product(left, right)
        out_mv, out_s = self.linear_out(gp, scalars=scalars)
        out_mv = self.norm(out_mv)
        return out_mv, out_s
```

"""Spin(3)-equivariant Geometric MLP for Cl(3,0)."""

from typing import List, Optional, Tuple, Union

import torch
from torch import nn

from su2eqnn.layers.dropout import GradeDropout
from su2eqnn.layers.linear import EquiLinear
from su2eqnn.layers.mlp.config import MLPConfig
from su2eqnn.layers.mlp.geometric_bilinears import GeometricBilinear
from su2eqnn.layers.mlp.nonlinearities import ScalarGatedNonlinearity


class GeoMLP(nn.Module):
    """Geometric MLP for Cl(3,0) dual-stream (MV + scalar).

    First layer: GeometricBilinear (GP-based equivariant nonlinearity).
    Subsequent layers: ScalarGatedNonlinearity → EquiLinear.
    Optional GradeDropout after first layer.

    Parameters
    ----------
    config : MLPConfig
        mv_channels : [in, h1, ..., out] — at least 2 entries
        s_channels  : [in, h1, ..., out] or None — same length as mv_channels
        activation  : gated activation for nonlinear layers
        dropout_prob : dropout probability after GeometricBilinear, or None
    """

    def __init__(self, config: MLPConfig) -> None:
        super().__init__()
        assert config.mv_channels is not None and len(config.mv_channels) >= 2, \
            "mv_channels must have at least 2 entries (in, out)"

        s_channels: List[Optional[int]] = (
            [None] * len(config.mv_channels)
            if config.s_channels is None
            else list(config.s_channels)
        )

        layers: List[nn.Module] = []

        # First layer: GeometricBilinear (equivariant nonlinearity via GP)
        layers.append(GeometricBilinear(
            in_mv_channels=config.mv_channels[0],
            out_mv_channels=config.mv_channels[1],
            in_s_channels=s_channels[0],
            out_s_channels=s_channels[1],
        ))

        # Optional dropout after first layer
        if config.dropout_prob is not None:
            layers.append(GradeDropout(config.dropout_prob))

        # Subsequent layers: gated activation → EquiLinear
        for in_mv, out_mv, in_s, out_s in zip(
            config.mv_channels[1:-1],
            config.mv_channels[2:],
            s_channels[1:-1],
            s_channels[2:],
        ):
            layers.append(ScalarGatedNonlinearity(config.activation))
            layers.append(EquiLinear(in_mv, out_mv, in_s_channels=in_s, out_s_channels=out_s))

        self.layers = nn.ModuleList(layers)

    def forward(
        self,
        multivectors: torch.Tensor,
        scalars: Optional[torch.Tensor] = None,
    ) -> Tuple[torch.Tensor, Union[torch.Tensor, None]]:
        """Forward pass.

        Parameters
        ----------
        multivectors : torch.Tensor with shape (..., mv_channels[0], 8)
        scalars      : torch.Tensor with shape (..., s_channels[0]) or None

        Returns
        -------
        out_mv : torch.Tensor with shape (..., mv_channels[-1], 8)
        out_s  : torch.Tensor with shape (..., s_channels[-1]) or None
        """
        mv, s = multivectors, scalars
        for layer in self.layers:
            mv, s = layer(mv, s)
        return mv, s
